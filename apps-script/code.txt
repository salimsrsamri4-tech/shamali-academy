var SPREADSHEET_ID = '16Qc1DUjhpbT-kIra74YeFQxzVxvBY7tSquUrHdacZKI';
var DRIVE_FOLDER_ID = '1AEsvtfftHBE6weOeLsqNoUr6LuHDYOnx';

// الأوراق التي تُفصل بين الأكاديميات بعمود "الأكاديمية"
var ACAD_COL = 'الأكاديمية';
var ACAD_SHEETS = ['بيانات المدربين','أرقام المدربين','تقويم الأكاديمية','المقترحات والملاحظات',
                   'تقارير المدربين','طلبات التقارير','المباريات','نتائج المباريات','جدول التدريب'];

function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, sheetName) {
  var sh = ss.getSheetByName(sheetName);
  if (!sh) sh = ss.insertSheet(sheetName);
  return sh;
}

function _ensureAcadCol(sh) {
  var lc = sh.getLastColumn();
  if (lc < 1 || sh.getLastRow() < 1) return;
  var headers = sh.getRange(1, 1, 1, lc).getValues()[0];
  if (headers.indexOf(ACAD_COL) < 0) sh.getRange(1, lc + 1).setValue(ACAD_COL);
}

function handleAppend(ss, sheetName, rowData) {
  var sh = getOrCreateSheet(ss, sheetName);
  if (sh.getLastRow() === 0) {
    var hs = Object.keys(rowData);
    sh.appendRow(hs);
    sh.getRange(1,1,1,hs.length).setFontWeight('bold');
  }
  if (rowData[ACAD_COL] !== undefined) _ensureAcadCol(sh);
  // أضف أي عمود جديد موجود في البيانات ولم يكن في صف العناوين بعد، بدل تجاهل قيمته
  var headers = sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0];
  var missing = Object.keys(rowData).filter(function(k){ return headers.indexOf(k) === -1; });
  if (missing.length) {
    sh.getRange(1, headers.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold');
    headers = headers.concat(missing);
  }
  sh.appendRow(headers.map(function(h){ return rowData[h]!==undefined ? rowData[h] : ''; }));
  return json({status:'ok'});
}
function doGet(e) {
  var p = e.parameter;
  var action = p.action || 'get';
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheetName = p.sheet || 'Sheet1';

  if (action === 'get') return _actGet(ss, sheetName, p);
  if (action === 'login') return _login(p);
  if (action === 'append') return _guardAppend(ss, sheetName, JSON.parse(p.data || '{}'), p.token);
  if (action === 'update') return _guardUpdate(ss, sheetName, p.column, p.value, JSON.parse(p.data || '{}'), p.token);
  if (action === 'delete') return _guardDelete(ss, sheetName, p.idCol || p.column || 'id', p.idVal !== undefined ? p.idVal : p.value, p.token);
  if (action === 'parentLookup') return _parentLookup(p);
  if (action === 'regCheck') return _regCheck(p);
  return json({status:'error', message:'Unknown action'});
}

function _actGet(ss, sheetName, p) {
  var sh = getOrCreateSheet(ss, sheetName);
  if (ACAD_SHEETS.indexOf(sheetName) > -1) _ensureAcadCol(sh);
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return json([]);
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var row = {};
    headers.forEach(function(h,j){ row[h] = data[i][j]; });
    rows.push(row);
  }
  rows = _redact(sheetName, rows, p.token);
  if (p.acad) rows = _acadFilter(sheetName, rows, String(p.acad).trim());
  var lim = parseInt(p.limit) || rows.length;
  // عند تجاوز الحد تُرجع الأحدث (آخر الصفوف) بدل الأقدم
  if (rows.length > lim) rows = rows.slice(rows.length - lim);
  return json(rows);
}

function _actUpdate(ss, sheetName, column, value, updateData) {
  var sh = getOrCreateSheet(ss, sheetName);
  var data = sh.getDataRange().getValues();
  var headers = data[0];
  var colIdx = headers.indexOf(column);
  if (colIdx === -1) return json({status:'error', message:'Column not found'});
  var missing = Object.keys(updateData).filter(function(k){ return headers.indexOf(k) === -1; });
  if (missing.length) {
    sh.getRange(1, headers.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold');
    headers = headers.concat(missing);
  }
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][colIdx]) === String(value)) {
      Object.keys(updateData).forEach(function(key) {
        var idx = headers.indexOf(key);
        if (idx !== -1) sh.getRange(i+1, idx+1).setValue(updateData[key]);
      });
      return json({status:'ok'});
    }
  }
  return json({status:'error', message:'Row not found'});
}

// حذف صف؛ إذا أُرسل اسم أكاديمية لا يُحذف إلا صفها
function _actDelete(ss, sheetName, column, value, acad) {
  var sh = getOrCreateSheet(ss, sheetName);
  var data = sh.getDataRange().getValues();
  var headers = data[0] || [];
  var colIdx = headers.indexOf(column);
  if (colIdx === -1) return json({status:'error', message:'Column not found'});
  var acadIdx = headers.indexOf(ACAD_COL);
  acad = String(acad || '').trim();
  for (var i = data.length - 1; i >= 1; i--) {
    if (String(data[i][colIdx]) !== String(value)) continue;
    if (acad && acadIdx > -1) {
      var rowAcad = String(data[i][acadIdx] || '').trim();
      if (rowAcad !== acad && !(rowAcad === '' && acad === BUILTIN_PASS.shamali.name)) continue;
    }
    sh.deleteRow(i + 1);
    return json({status:'ok'});
  }
  return json({status:'error', message:'Row not found'});
}
function _digits(v) {
  return String(v == null ? '' : v)
    .replace(/[٠-٩]/g, function(d){ return d.charCodeAt(0) - 1632; })
    .replace(/\D/g, '');
}

function _sheetObjects(name) {
  var sh = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(name);
  if (!sh) return [];
  var v = sh.getDataRange().getValues();
  var h = v[0] || [];
  return v.slice(1).map(function(row){
    var o = {};
    h.forEach(function(k, i){ o[k] = row[i]; });
    return o;
  });
}

// بوابة ولي الأمر: أبناء هذا الرقم المدني فقط
function _parentLookup(p) {
  var civil = _digits(p.civil);
  if (civil.length < 6) return json({status:'ok', v:1, data:[], photos:[]});
  var kids = _sheetObjects('الورقة1').filter(function(r){
    return _digits(r['الرقم المدني']) === civil;
  });
  var ids = {};
  kids.forEach(function(r){ ids[String(r['رقم الملف'])] = 1; });
  var photos = _sheetObjects('صور اللاعبين').filter(function(r){
    return ids[String(r['رقم الملف'])];
  });
  return json({status:'ok', v:1, data:kids, photos:photos});
}

// تسجيل لاعب: هل الهوية مكررة + رقم ملف جديد غير مستخدم
function _regCheck(p) {
  var pid = _digits(p.pid);
  var used = {}, exists = false;
  _sheetObjects('الورقة1').forEach(function(r){
    used[String(r['رقم الملف']).trim()] = 1;
    if (pid && _digits(r['رقم هوية اللاعب']) === pid) exists = true;
  });
  var newId;
  do { newId = 'SH-' + Math.floor(10000 + Math.random() * 90000); } while (used[newId]);
  return json({status:'ok', v:1, exists:exists, newId:newId});
}
function doPost(e) {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);

  var body;
  try {
    body = JSON.parse(e.postData.contents || '{}');
  } catch(err) {
    body = {
      action: e.parameter.action || 'append',
      sheet:  e.parameter.sheet  || 'Sheet1',
      data:   JSON.parse(e.parameter.data || '{}')
    };
  }

  var action = body.action || '';
  var sheetName = body.sheet || 'Sheet1';

  if (action === 'append') return _guardAppend(ss, sheetName, body.data || {}, body.token);
  if (action === 'update') return _guardUpdate(ss, sheetName, body.column, body.value, body.data || {}, body.token);
  if (action === 'delete') {
    var val = body.value !== undefined ? body.value : body.idVal;
    return _guardDelete(ss, sheetName, body.column || body.idCol || 'id', val, body.token);
  }
  if (action === 'uploadFile') return _uploadOne(body);
  if (action === 'uploadFiles' || (!action && (body.parentIdFile || body.playerPhoto || body.playerDoc))) {
    return _uploadMany(body);
  }
  return json({status:'error', message:'Unknown action'});
}

// رفع ملف واحد (من نموذج تسجيل الأكاديمية)
function _uploadOne(body) {
  try {
    var folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    var decoded = Utilities.base64Decode(body.fileData || '');
    if (!_okMime(body.mimeType || 'image/jpeg')) return json({status:'error', message:'نوع الملف غير مسموح'});
    var blob = Utilities.newBlob(decoded, body.mimeType || 'image/jpeg', body.fileName || 'file');
    var file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    var fileId = file.getId();
    var fileUrl = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w400';
    return json({status:'ok', fileUrl: fileUrl, fileId: fileId});
  } catch(err) {
    return json({status:'error', message: err.toString()});
  }
}

// رفع ملفات متعددة (من نموذج تسجيل اللاعبين)
function _uploadMany(body) {
  try {
    var folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    var result = {status: 'success', files: {}};
    ['parentIdFile', 'playerPhoto', 'playerDoc'].forEach(function(key) {
      if (!body[key]) return;
      try {
        var raw = body[key];
        var base64Data = raw.indexOf(',') > -1 ? raw.split(',')[1] : raw;
        var mimeType = raw.indexOf('data:') === 0 ? raw.split(';')[0].split(':')[1] : 'image/jpeg';
        if (!_okMime(mimeType)) { result.files[key + 'Id'] = ''; return; }
        var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), mimeType, key + '_' + (body.name || Date.now()));
        var file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        result.files[key + 'Id'] = 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w400';
      } catch(e) {
        result.files[key + 'Id'] = '';
      }
    });
    return json(result);
  } catch(err) {
    return json({status:'error', message: err.toString()});
  }
}

// ===== الحماية: من يحق له الإضافة والتعديل والحذف =====
// تحتاج من Auth.gs: _session, BUILTIN_PASS, PASS_COLS, JOIN_SHEET, _acadKey

// نماذج يملؤها الزوار بدون دخول
var PUBLIC_APPEND = ['الورقة1', 'أكاديميات المنصة', 'المقترحات والملاحظات', 'طلبات التقارير',
  'تجديدات الاشتراكات', 'شكاوى البطولة', 'لاعبو البطولة', 'الأكاديميات المسجلة', 'طلبات اللاعبين',
  'حذف اللاعبين', 'تسجيل مهرجان نبض المستقبل 2026', 'الجهاز الفني والإداري - مهرجان نبض المستقبل 2026'];
var PLAYER_SHEETS = ['سجل الحضور', 'سجل الحضور اليومي', 'تجديدات الاشتراكات', 'تقييمات اللاعبين',
  'صور اللاعبين', 'تكريم المجيدين'];
var DENY_SHEET = 'سجل الرفض';

function _okMime(m) { return /^image\//.test(String(m)) || String(m) === 'application/pdf'; }

// تنقية النصوص من الأكواد قبل حفظها (كلمات المرور تُترك كما هي)
function _cleanRow(row) {
  var out = {};
  Object.keys(row || {}).forEach(function(k) {
    var v = row[k];
    if (typeof v === 'string' && PASS_COLS.indexOf(k) < 0) {
      v = v.replace(/[<>]/g, '').replace(/"/g, '”').replace(/'/g, '’');
      if (/^\s*(javascript|vbscript|data):/i.test(v)) v = '';
    }
    out[k] = v;
  });
  return out;
}

function _deny(action, sheetName, reason) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sh = getOrCreateSheet(ss, DENY_SHEET);
    if (sh.getLastRow() === 0) sh.appendRow(['الوقت', 'العملية', 'الورقة', 'السبب']);
    sh.appendRow([new Date(), action, sheetName, reason]);
  } catch (e) {}
  return json({status:'error', message: reason});
}

function _guardAppend(ss, sheetName, rowData, token) {
  if (Array.isArray(rowData)) {
    var res = json({status:'ok'});
    rowData.forEach(function(r) { res = _guardAppend(ss, sheetName, r || {}, token); });
    return res;
  }
  var s = _session(token);
  if (!s && PUBLIC_APPEND.indexOf(sheetName) < 0) return _deny('إضافة', sheetName, 'تحتاج تسجيل دخول');
  if (!s || s.role !== 'super') {
    rowData = _cleanRow(rowData);
    if (sheetName === JOIN_SHEET) {
      rowData['الحالة'] = 'جديد';
      var key = String(rowData['مفتاح_الأكاديمية'] || '').trim();
      if (key && _sheetObjects(JOIN_SHEET).some(function(r) { return _acadKey(r) === key; }))
        return _deny('إضافة', sheetName, 'مفتاح أكاديمية مكرر');
    }
    if (s && s.name && ACAD_SHEETS.indexOf(sheetName) > -1) rowData[ACAD_COL] = s.name;
  }
  return handleAppend(ss, sheetName, rowData);
}

function _guardUpdate(ss, sheetName, column, value, data, token) {
  var s = _session(token);
  if (!s) return _deny('تعديل', sheetName, 'تحتاج تسجيل دخول');
  if (s.role === 'super') return _actUpdate(ss, sheetName, column, value, data);
  if (s.role !== 'admin') return _deny('تعديل', sheetName, 'صلاحية المدير مطلوبة');
  data = _cleanRow(data);
  if (sheetName === JOIN_SHEET) {
    if (column !== 'مفتاح_الأكاديمية' || String(value) !== String(s.acad))
      return _deny('تعديل', sheetName, 'لا يمكن تعديل أكاديمية أخرى');
    ['الحالة', 'مفتاح_الأكاديمية', 'رقم الطلب', 'اسم الأكاديمية'].forEach(function(k) { delete data[k]; });
  }
  return _actUpdate(ss, sheetName, column, value, data);
}

function _guardDelete(ss, sheetName, column, value, token) {
  var s = _session(token);
  if (!s) return _deny('حذف', sheetName, 'تحتاج تسجيل دخول');
  if (s.role === 'super') return _actDelete(ss, sheetName, column, value, '');
  if (s.role !== 'admin') return _deny('حذف', sheetName, 'صلاحية المدير مطلوبة');
  if (sheetName === JOIN_SHEET) return _deny('حذف', sheetName, 'غير مسموح');
  var sh = ss.getSheetByName(sheetName);
  var lc = sh ? sh.getLastColumn() : 0;
  var hasAcad = lc > 0 && sh.getRange(1, 1, 1, lc).getValues()[0].indexOf(ACAD_COL) > -1;
  if (!hasAcad && s.acad !== 'shamali') return _deny('حذف', sheetName, 'لا يمكن الحذف من هذه الورقة');
  return _actDelete(ss, sheetName, column, value, hasAcad ? s.name : '');
}

// تقليل البيانات المرسلة: صفوف أكاديمية واحدة فقط (الحماية الفعلية في _redact والحراس أعلاه)
function _acadMatch(rowAcad, acad) {
  rowAcad = String(rowAcad || '').trim();
  return rowAcad === acad || (rowAcad === '' && acad === BUILTIN_PASS.shamali.name);
}

function _acadFilter(sheetName, rows, acad) {
  if (sheetName === 'الورقة1' || ACAD_SHEETS.indexOf(sheetName) > -1) {
    if (!rows.length || !(ACAD_COL in rows[0])) return rows;
    return rows.filter(function(r) { return _acadMatch(r[ACAD_COL], acad); });
  }
  if (PLAYER_SHEETS.indexOf(sheetName) > -1) {
    var ids = {};
    _sheetObjects('الورقة1').forEach(function(r) {
      if (_acadMatch(r[ACAD_COL], acad)) ids[String(r['رقم الملف']).trim()] = 1;
    });
    var isShamali = acad === BUILTIN_PASS.shamali.name;
    return rows.filter(function(r) {
      var id = String(r['رقم الملف'] || r['ID'] || r['معرف اللاعب'] || '').trim();
      return id ? !!ids[id] : isShamali;
    });
  }
  return rows;
}
